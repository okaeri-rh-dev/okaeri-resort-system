# Phase 2a: Types and Constants

## Purpose

This document defines the TypeScript types, enums, and constants required for the audit trail system. These types extend the existing `database.types.ts` file and provide type-safe interfaces for logging, querying, and displaying audit data.

**Key Principle**: Extend existing patterns, don't create parallel type systems.

## Type Definitions

### File Location

Add these types to your existing `database.types.ts` file (generated by Supabase, but extended manually):

```typescript

/**
 * Audit log action types
 * Maps to PostgreSQL audit trigger actions
 */
export type AuditAction = "INSERT" | "UPDATE" | "DELETE";

/**
 * Application-level audit categories
 * Used for higher-level business context (Layer 2)

export enum AuditCategory {
  AUTHENTICATION = "authentication",
  BOOKING = "booking",

/**
 * Severity levels for audit events
 * Helps prioritize which logs to review
 * NOTE: MUST be uppercase to match database CHECK constraint

  INFO = "INFO", // Normal operations (booking created, unit updated)
  WARNING = "WARNING", // Unusual but valid (booking cancelled, payment refund)
  CRITICAL = "CRITICAL", // Requires attention (deletion, security event)
  DEBUG = "DEBUG", // Verbose debugging information
}

/**
 * Audit log record (matches database schema exactly)
 * All fields align with audit_logs table columns
 */
export interface AuditLog {
  audit_id: string; // UUID primary key
  correlation_id: string; // UUID for linking related log entries across layers
  table_name: string; // Table affected (bookings, units, system, authentication, etc)
  record_id: string | null; // TEXT type identifier (supports UUID and non-UUID PKs)
  record_pk: Record<string, unknown> | null; // Composite primary key as JSON
  action: AuditAction; // INSERT, UPDATE, DELETE
  event_type: string | null; // Business event type for app events (e.g., 'login', 'cancel_booking')
  source: "db" | "app"; // Layer 1 (db triggers) or Layer 2 (app logging)

  // User context (from Supabase Auth)
  user_id: string | null; // Supabase Auth user ID
  user_email: string | null; // User email for display
  user_role: string | null; // User role at time of action (owner/manager/staff)

  // Data capture (Layer 1: triggers)
  old_data: Record<string, unknown> | null; // JSONB before state
  new_data: Record<string, unknown> | null; // JSONB after state
  changed_fields: string[] | null; // Array of field names changed (UPDATE only)

  // Business context (Layer 2: app logging)
  description: string | null; // Human-readable description
  category: string | null; // Business category (authentication, booking, etc)
  severity: "INFO" | "WARNING" | "CRITICAL" | "DEBUG" | null; // Severity level (UPPERCASE)
  metadata: Record<string, unknown> | null; // Additional structured context

  // Client context (Layer 2: optional)
  ip_address: string | null; // Client IP address (often NULL from browser)
  user_agent: string | null; // Browser user agent string

  // Timing
  created_at: string; // ISO 8601 timestamp (CRITICAL: use created_at, NOT timestamp)
}

/**
 * Filter options for querying audit logs
 */
export interface AuditLogFilters {
  // Date range
  startDate?: string; // ISO 8601 date
  endDate?: string; // ISO 8601 date

  // Entity filters
  userEmail?: string; // Filter by specific user
  userId?: string; // Filter by Supabase Auth ID
  tableName?: string; // Filter by table (bookings, units, etc)
  recordId?: string; // Filter by specific record UUID
  correlationId?: string; // Filter by correlation ID (find all related logs)
  action?: AuditAction; // Filter by operation type
  source?: "db" | "app"; // Filter by source (database triggers vs app logging)

  // Business context (Layer 2 logging)
  eventType?: string; // Filter by business event type (e.g., 'login', 'cancel_booking')
  category?: AuditCategory; // Filter by business category
  severity?: AuditSeverity; // Filter by severity level

  // Pagination
  limit?: number; // Records per page (default 50)
  offset?: number; // Skip N records (for pagination)
}

/**
 * Query result with pagination metadata
 */
export interface AuditLogQueryResult {
  data: AuditLog[];
  totalCount: number;
  hasMore: boolean;
  currentPage: number;
  totalPages: number;
}

/**
 * Statistics aggregation result
 */
export interface AuditLogStats {
  totalLogs: number;
  byTable: Record<string, number>; // { bookings: 150, units: 30, ... }
  byAction: Record<AuditAction, number>; // { INSERT: 100, UPDATE: 80, DELETE: 5 }
  byUser: Array<{ email: string; count: number }>;
  recentActivity: Array<{ date: string; count: number }>; // Last 7 days
}

/**
 * Context for creating audit logs from application
 * Used with logAuditEvent() function (Layer 2)
 *
 * IMPORTANT: Layer 2 logs always use action='INSERT' and source='app' internally.
 * Do NOT include an 'action' field in this context - it's set automatically.
 *
 * For semantic action meaning, use:
 * - eventType: Business event type (e.g., AuditEventType.CANCEL_BOOKING)
 * - description: Human-readable action (e.g., "Booking cancelled by guest")
 * - metadata.action: Additional action context if needed
 */
export interface AuditEventContext {
  tableName: string;
  recordId?: string;
  recordPk?: Record<string, unknown>; // Composite primary key support (for future use)
  correlationId?: string; // UUID to link this log with related trigger logs
  eventType?: AuditEventType; // Optional: semantic business event type (e.g., AuditEventType.CANCEL_BOOKING)
  description: string; // Human-readable description (e.g., "Booking cancelled by guest")
  category: AuditCategory;
  severity?: AuditSeverity; // Default: INFO
  metadata?: Record<string, unknown>; // Business-specific data
  oldData?: Record<string, unknown>; // Optional: before state
  newData?: Record<string, unknown>; // Optional: after state
}

/**
 * Export format options
 */
export type AuditExportFormat = "csv" | "json";

export interface AuditExportOptions {
  format: AuditExportFormat;
  filters?: AuditLogFilters;
  includeMetadata?: boolean; // Include JSONB fields in export
  filename?: string; // Custom filename (default: audit-logs-{date}.csv)
}
```

## Constants

### Sensitive Fields Configuration

Create a constants file: `src/lib/audit-constants.ts`

```typescript
// src/lib/audit-constants.ts

/**
 * SENSITIVE_FIELDS - Fields that should be redacted in audit logs
 *
 * Why redaction?
 * - Guest privacy: PII shouldn't be in plain text logs
 * - Security: Payment details need extra protection
 * - Compliance: Even internal logs should minimize PII exposure
 *
 * What's redacted?
 * - Guest personal information (phone, email, address)
 * - Payment sensitive data (card numbers, GCash reference)
 * - Authentication secrets (passwords, tokens)
 */
export const SENSITIVE_FIELDS: Record<string, string[]> = {
  // Guest PII
  guests: [
    "guest_phone",
    "guest_email",
    "guest_address",
    "guest_notes", // May contain sensitive personal info
    "emergency_contact_phone",
  ],

  // Booking PII (guest details captured in bookings)
  bookings: [
    "special_requests", // May contain medical/dietary info
    "internal_notes", // Staff notes may contain sensitive context
  ],

  // Payment sensitive data
  payments: [
    "payment_method_details", // GCash reference numbers, bank details
    "transaction_reference", // External payment IDs
    "receipt_url", // May expose sensitive file paths
  ],

  // User authentication (if you add user management later)
  users: ["password_hash", "reset_token", "email_verification_token"],

  // Expense receipts (may contain personal vendor info)
  expenses: ["vendor_contact", "receipt_url"],
};

/**
 * Get sensitive fields for a specific table
 * @param tableName - Table to check
 * @returns Array of field names to redact
 */
export function getSensitiveFields(tableName: string): string[] {
  return SENSITIVE_FIELDS[tableName] || [];
}

/**
 * Check if a field should be redacted
 * @param tableName - Table name
 * @param fieldName - Field to check
 * @returns true if field should be redacted
 */
export function isSensitiveField(tableName: string, fieldName: string): boolean {
  const sensitiveFields = getSensitiveFields(tableName);
  return sensitiveFields.includes(fieldName);
}

/**
 * REDACTION_PLACEHOLDER - Value used for redacted fields
 */
export const REDACTION_PLACEHOLDER = "[REDACTED]";

/**
 * Redact sensitive fields in an object
 * @param tableName - Table name
 * @param data - Data object to redact
 * @returns New object with sensitive fields redacted
 */
export function redactSensitiveData(
  tableName: string,
  data: Record<string, unknown> | null
): Record<string, unknown> | null {
  if (!data) return null;

  const sensitiveFields = getSensitiveFields(tableName);
  if (sensitiveFields.length === 0) return data;

  const redacted = { ...data };
  sensitiveFields.forEach((field) => {
    if (field in redacted) {
      redacted[field] = REDACTION_PLACEHOLDER;
    }
  });

  return redacted;
}
```

### Table Name Constants

```typescript
// src/lib/audit-constants.ts (continued)

/**
 * AUDITED_TABLES - All tables with audit triggers (Layer 1)
 * Plus special table names for app events (Layer 2)
 */
export const AUDITED_TABLES = [
  // Physical database tables (Layer 1: triggers attached)
  "units",
  "guests",
  "bookings",
  "payments",
  "expenses",
  "maintenance_logs",

  // Virtual tables for app events (Layer 2: no triggers, used for categorization)
  "system", // System-wide events (migrations, config changes)
  "authentication", // Auth events (login, logout, failed attempts)
] as const;

export type AuditedTable = (typeof AUDITED_TABLES)[number];

/**
 * Validate if a table name is valid for audit logging
 * Accepts both physical tables (with triggers) and virtual tables (for app events)
 */
export function isAuditedTable(tableName: string): tableName is AuditedTable {
  return AUDITED_TABLES.includes(tableName as AuditedTable);
}

/**
 * Check if a table has database triggers (Layer 1)
 */
export function hasAuditTrigger(tableName: string): boolean {
  const TRIGGER_TABLES = [
    "units",
    "guests",
    "bookings",
    "payments",
    "expenses",
    "maintenance_logs",
  ];
  return TRIGGER_TABLES.includes(tableName);
}

 * AUDIT_PSEUDO_TABLES - Virtual tables for application events
```

### Owner Email Utility

```typescript
// src/lib/audit-constants.ts (continued)

/**
 * Get list of owner emails from environment
 */
export function getOwnerEmails(): string[] {
  const ownerEmailsEnv = import.meta.env.VITE_OWNER_EMAILS;
  if (!ownerEmailsEnv) {
    console.warn("VITE_OWNER_EMAILS not configured. Audit logs will have no owners.");
    return [];
  }
  return ownerEmailsEnv.split(",").map((email) => email.trim());
}

/**
 * Check if an email is an owner
 */
export function isOwnerEmail(email: string): boolean {
  const owners = getOwnerEmails();
  return owners.includes(email);
}

/**
 * Debug flag for audit system
 */
export const AUDIT_DEBUG = import.meta.env.VITE_AUDIT_DEBUG === "true";

export function auditDebugLog(message: string, data?: unknown): void {
  if (AUDIT_DEBUG) {
    console.log(`[AUDIT DEBUG] ${message}`, data);
  }
}
```
